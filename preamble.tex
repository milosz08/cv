% #1: name, #2: localization, #3: title/role, #4: date
\newcommand{\resumeSubheading}[4]{
  \item
    \vspace{-2pt}
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{\small #1} & \textit{\small #2} \\
      \small #3 & \textit{\small #4} \\
    \end{tabular*}
    \vspace{-2pt}
}

% #1: title, #2: date, #3: description/content
\newcommand{\resumeGeneralItem}[3]{
  \item
    \vspace{-2pt}
    \begin{tabularx}{0.97\textwidth}[t]{@{}X >{\raggedleft\arraybackslash}p{80pt}@{}}
      \textbf{\small #1} & \textit{\small #2} \\
      \multicolumn{2}{@{}p{0.97\textwidth}@{}}{\small #3}
    \end{tabularx}
    \vspace{-2pt}
}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
